#!/usr/bin/env python3
"""Bound admission through owned cleanup, without changing the reviewed core runner."""
import datetime
import hashlib
import json
import os
import pathlib
import signal
import subprocess
import sys
import time

RUNNER = '/tmp/ux-610-evidence/proof/run-ticket-stabilized.py'

def parsed(rows):
    return [row.split(None, 3) for row in rows if len(row.split(None, 3)) == 4]

def supervise(process, *, root, phase, label, limit, started, clock, inspect_rows,
              group_signal, receipt_group, interrupted=lambda: None,
              simulator_inventory, simulator_command, save_record, simulator_name, initial_process_ids, native_phase):
    errors, notes = [], []
    worker, worker_observed = None, False
    deadline_stop = False

    def problem(stage, error):
        CLEANUP.issue(errors, stage, error)

    def rows():
        return parsed(inspect_rows())

    def group_rows(group):
        return [" ".join(row) for row in rows() if row[2] == str(group)]

    def discover():
        nonlocal worker, worker_observed
        found = [int(pid) for pid, ppid, pgid, command in rows()
                 if pid == pgid and int(pid) not in initial_process_ids
                 and command.split() == [sys.executable, "-u", RUNNER, phase, label, "--worker"]]
        if len(found) > 1 or (found and worker is not None and found[0] != worker):
            raise RuntimeError("Ambiguous or changed exact worker identity")
        if found:
            worker, worker_observed = found[0], True

    try:
        while process.poll() is None and clock() - started < limit - 60:
            if interrupted():
                raise RuntimeError("Interrupted: " + str(interrupted()))
            discover()
            try:
                process.wait(timeout=min(5, max(.01, limit - 60 - (clock() - started))))
            except subprocess.TimeoutExpired:
                pass
        if process.poll() is None:
            deadline_stop = True
    except BaseException as error:
        problem("outer monitor", error)
    finally:
        try:
            if process.poll() is None:
                process.send_signal(signal.SIGTERM)
                process.wait(timeout=max(.01, min(45, limit - 10 - (clock() - started))))
        except BaseException as error:
            problem("owned supervisor graceful stop", error)
        supervisor_after = None
        try:
            drained = CLEANUP.drain_group(process.pid, process, group_rows, group_signal, clock=clock)
            errors.extend(drained["errors"])
            supervisor_after = drained["members_after"]
            process.wait(timeout=3)
        except BaseException as error:
            problem("owned supervisor group cleanup", error)
        try:
            discover()
        except BaseException as error:
            problem("owned worker final discovery", error)
        try:
            recorded = receipt_group()
            if recorded is not None:
                assert isinstance(recorded, int) and recorded > 1 and recorded not in initial_process_ids, "Invalid or preexisting worker group"
            if worker is not None and recorded is not None and worker != recorded:
                raise RuntimeError("Core group differs from exact observed worker")
            if worker is None:
                worker = recorded
        except BaseException as error:
            problem("owned worker record recovery", error)
        worker_after = None
        try:
            if worker is None:
                raise RuntimeError("Worker identity unknown; no guessed group signal")
            before = group_rows(worker)
            scoped = worker_observed or any(str(root) in row for row in before)
            if before and not scoped:
                raise RuntimeError("Unproven recovered group; no unrelated signal")
            drained = CLEANUP.drain_group(worker, None, group_rows, group_signal, clock=clock)
            errors.extend(drained["errors"])
            worker_after = drained["members_after"]
        except BaseException as error:
            problem("owned worker independent cleanup", error)
        state = {}
        try:
            state = json.loads((pathlib.Path(root) / (label + "-receipt.json")).read_text())
        except BaseException as error:
            problem("core final receipt recovery", error)
        simulator = {"owned_simulator_deleted": False}
        if native_phase:
            try:
                simulator = CLEANUP.cleanup_simulator(
                    root, label, simulator_name, state, simulator_inventory, simulator_command, save_record,
                )
                errors.extend(simulator["cleanup_errors"])
            except BaseException as error:
                problem("independent owned UUID cleanup", error)
        try:
            final_rows = rows()
            supervisor_after = [" ".join(row) for row in final_rows if row[2] == str(process.pid)]
            if worker is not None:
                worker_after = [" ".join(row) for row in final_rows if row[2] == str(worker)]
            known = True
        except BaseException as error:
            problem("independent final group absence", error)
            supervisor_after = worker_after = None
            known = False
        try:
            runner_rc = process.poll()
        except BaseException as error:
            problem("final supervisor poll", error)
            runner_rc = None
        if interrupted():
            problem("outer interrupted", RuntimeError(str(interrupted())))
        elapsed = clock() - started
        report = dict(
            phase=phase, label=label, runner_pid=process.pid, worker_group=worker,
            runner_rc=runner_rc, deadline_stop=deadline_stop, errors=errors, notes=notes,
            outer_started_monotonic=started, outer_duration_seconds=elapsed, outer_bound_seconds=limit,
            closed_at=datetime.datetime.now(datetime.timezone.utc).isoformat(),
            final_process_inspection_known=known, owned_worker_members_after=worker_after,
            owned_supervisor_members_after=supervisor_after, independent_simulator_cleanup=simulator,
        )
        simulator_closed = not native_phase or simulator.get("owned_simulator_deleted") is True or simulator.get("owned_simulator_not_created") is True
        report["rc"] = 0 if (
            runner_rc == state.get("rc") == 0 and not deadline_stop and not errors and elapsed < limit
            and worker_after == supervisor_after == [] and known and simulator_closed
        ) else 1
    return report

if __name__ == "__main__":
    raise SystemExit("Library only: use the exact reviewed adapter.")
